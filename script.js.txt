/* =========================
   FORMATAÇÃO
========================= */

function dinheiro(valor) {

    return valor.toLocaleString(
        "pt-BR",
        {
            style: "currency",
            currency: "BRL"
        }
    );

}


/* =========================
   CRIAR PRODUTOS
========================= */

function carregarProdutos() {

    Object.keys(produtos).forEach(categoria => {

        const container =
            document.getElementById(
                "lista-" + categoria
            );

        if (!container) return;

        produtos[categoria].forEach(produto => {

            const nome = produto[0];

            const preco = produto[1];

            const descricao = produto[2];

            const card =
                document.createElement("article");

            card.className = "card";

            card.innerHTML = `

                <h3>${nome}</h3>

                ${
                    descricao
                    ?
                    `<p>${descricao}</p>`
                    :
                    ""
                }

                <div class="preco">
                    ${dinheiro(preco)}
                </div>

                <button
                    class="adicionar"
                    onclick="adicionarProduto('${nome}', ${preco}, '${categoria}')">

                    🛒 ADICIONAR AO PEDIDO

                </button>

            `;

            container.appendChild(card);

        });

    });

}


/* =========================
   ADICIONAR PRODUTO
========================= */

function adicionarProduto(nome, preco, categoria) {

    /*
       Lanches terão opções de adicionais.
    */

    if (categoria === "lanches") {

        abrirOpcoesLanche(nome, preco);

        return;

    }


    /*
       Baldes terão escolha do tipo de frango.
    */

    if (categoria === "baldes") {

        abrirOpcoesBalde(nome, preco);

        return;

    }


    adicionarCarrinho(
        nome,
        preco,
        ""
    );

}

function confirmarLanche(nome, preco) {

    const selecionados =
        document.querySelectorAll(
            "#modalLancheTemporario .adicional-lanche:checked"
        );

    let adicionais = [];

    let valorAdicionais = 0;


    selecionados.forEach(input => {

        const index =
            Number(input.value);

        const adicional =
            produtos.adicionais[index];

        adicionais.push(
            adicional[0]
        );

        valorAdicionais +=
            adicional[1];

    });


    let descricao =
        "Acompanha 1 molho de alho.";


    if (adicionais.length > 0) {

        descricao +=
            " Adicionais: " +
            adicionais.join(", ");

    }


    adicionarCarrinho(
        nome +
        (
            adicionais.length
            ?
            " (" +
            adicionais.join(", ") +
            ")"
            :
            ""
        ),
        preco + valorAdicionais,
        descricao
    );


    fecharModalLanche();

}

function confirmarBalde(nome, preco) {

    const escolhido =
        document.querySelector(
            "#modalBaldeTemporario input[name='tipoFrango']:checked"
        );


    if (!escolhido) {

        alert(
            "Escolha o tipo de frango."
        );

        return;

    }


    let valorFinal = preco;

    let descricaoFrango = "";


    if (escolhido.value === "Misto") {

        const misto =
            document.querySelectorAll(
                "#modalBaldeTemporario .tipo-misto:checked"
            );


        if (misto.length !== 2) {

            alert(
                "No Misto, escolha exatamente 2 tipos de frango."
            );

            return;

        }


        const tipos =
            Array.from(misto)
                .map(item => item.value);


        descricaoFrango =
            "Misto: " +
            tipos.join(" + ");

        valorFinal += 5;

    }

    else {

        descricaoFrango =
            escolhido.value;

    }


    const descricao =
        descricaoFrango +
        " — Acompanha 1 ketchup e 1 barbecue.";


    adicionarCarrinho(
        nome +
        " (" +
        descricaoFrango +
        ")",
        valorFinal,
        descricao
    );


    fecharModalBalde();

}


/* =========================
   CARRINHO
========================= */

function adicionarCarrinho(
    nome,
    preco,
    descricao = ""
) {

    const item =
        carrinho.find(
            produto =>
                produto.nome === nome &&
                produto.descricao === descricao
        );


    if (item) {

        item.quantidade++;

    }

    else {

        carrinho.push({

            nome: nome,

            preco: preco,

            quantidade: 1,

            descricao: descricao

        });

    }


    atualizarCarrinho();

}


/* =========================
   ATUALIZAR CARRINHO
========================= */

function atualizarCarrinho() {

    const container =
        document.getElementById(
            "itensCarrinho"
        );

    const contador =
        document.getElementById(
            "quantidadeCarrinho"
        );


    if (!container || !contador) return;


    const quantidade =
        carrinho.reduce(
            (total, item) =>
                total +
                item.quantidade,
            0
        );


    contador.textContent =
        quantidade;


    container.innerHTML = "";


    if (carrinho.length === 0) {

        container.innerHTML =
            "<p>Seu pedido está vazio. 🍗</p>";

    }


    carrinho.forEach(
        (item, index) => {

            const div =
                document.createElement(
                    "div"
                );

            div.className =
                "item-carrinho";


            div.innerHTML = `

                <div>

                    <strong>
                        ${item.quantidade}x
                        ${item.nome}
                    </strong>

                    ${
                        item.descricao
                        ?
                        `<small>${item.descricao}</small>`
                        :
                        ""
                    }

                    <br>

                    ${dinheiro(
                        item.preco *
                        item.quantidade
                    )}

                </div>

                <div class="controles">

                    <button
                        onclick="alterarQuantidade(${index}, -1)">

                        −

                    </button>

                    <button
                        onclick="alterarQuantidade(${index}, 1)">

                        +

                    </button>

                </div>

            `;


            container.appendChild(div);

        }
    );


    atualizarTotal();

}


/* =========================
   QUANTIDADE
========================= */

function alterarQuantidade(
    index,
    valor
) {

    carrinho[index].quantidade +=
        valor;


    if (
        carrinho[index].quantidade <= 0
    ) {

        carrinho.splice(
            index,
            1
        );

    }


    atualizarCarrinho();

}


/* =========================
   TOTAL
========================= */

function calcularSubtotal() {

    return carrinho.reduce(

        (total, item) =>

            total +
            item.preco *
            item.quantidade,

        0

    );

}


function atualizarTotal() {

    const subtotal =
        calcularSubtotal();


    const entrega =
        tipoRecebimento === "entrega"
        ?
        TAXA_ENTREGA
        :
        0;


    const total =
        subtotal + entrega;


    const subtotalElemento =
        document.getElementById(
            "subtotal"
        );

    const entregaElemento =
        document.getElementById(
            "valorEntrega"
        );

    const totalElemento =
        document.getElementById(
            "total"
        );


    if (subtotalElemento)
        subtotalElemento.textContent =
            dinheiro(subtotal);


    if (entregaElemento)
        entregaElemento.textContent =
            dinheiro(entrega);


    if (totalElemento)
        totalElemento.textContent =
            dinheiro(total);

}


/* =========================
   ENTREGA / RETIRADA
========================= */

function escolherEntrega(tipo) {

    tipoRecebimento =
        tipo;


    const entrega =
        document.getElementById(
            "botaoEntrega"
        );

    const retirada =
        document.getElementById(
            "botaoRetirada"
        );


    entrega.classList.remove(
        "selecionado"
    );

    retirada.classList.remove(
        "selecionado"
    );


    const campoEndereco =
        document.getElementById(
            "campoEndereco"
        );


    if (tipo === "entrega") {

        entrega.classList.add(
            "selecionado"
        );

        campoEndereco.style.display =
            "block";

    }

    else {

        retirada.classList.add(
            "selecionado"
        );

        campoEndereco.style.display =
            "none";

    }


    atualizarTotal();

}
